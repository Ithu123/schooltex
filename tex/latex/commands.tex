\newcommand{\getpagetotal}{\the\pagetotal}
\newcommand{\gettextheight}{\the\textheight}
\newcommand{\getbaselineskip}{\the\baselineskip}
%grid command
\ProvideDocumentCommand{\grid}{O{0}}{
\vspace{\baselineskip}

\pgfmathsetmacro{\theight}{scalar(\gettextheight)}
\pgfmathsetmacro{\pat}{scalar(\getpagetotal)}
\pgfmathsetmacro{\bsskip}{scalar(\getbaselineskip)}

\dimendef\gridwidth=0
\pgfmathsetlength{\gridwidth}{1 cm * floor(scalar(\textwidth)*0.0352778) }

\ifnum#1=0
  \pgfmathsetmacro{\gridheight}{1 cm * floor((\theight-\pat-\bsskip)*0.0352778) }
\else
  \pgfmathsetmacro{\gridheight}{1 cm *#1 }
\fi

  \tikz 
  \coordinate (A) at (canvas cs:x=\gridwidth,y=\gridheight) {};
  \draw[step=0.5cm, color=gray] (0cm ,0 cm ) grid (canvas cs:x=\gridwidth,y=\gridheight);
\end{tikzpicture}
}%

% Subfigure-Caption in Minipages (ohne figure-Float).
% Erster \minicaption-Aufruf einer Gruppe zieht den figure-Counter vor,
% damit \label die richtige Abbildungsnummer bekommt.
% Abschluss der Gruppe: \minimaincaption{Text} statt \captionof{figure}{...}.
%
% Verwendung:
%   \minicaption[label-key]{Text}   — Teilbild-Beschriftung (Label optional)
%   \minimaincaption{Text}          — Hauptbeschriftung (setzt LOF-Eintrag)
\makeatletter
\newif\if@minicapactive

\newcommand{\minicaption}[2][]{%
    \if@minicapactive\else
        \stepcounter{figure}%           figure-Nummer vorziehen
        \global\@minicapactivetrue
    \fi
    \def\@captype{subfigure}%
    \caption{#2}%
    \def\@tempa{#1}%
    \ifx\@tempa\@empty\else\label{#1}\fi%
}

\newcommand{\minimaincaption}[1]{%
    \addtocounter{figure}{-1}%          vorgezogenen Schritt rückgängig
    \captionof{figure}{#1}%             zählt hoch + LOF-Eintrag
    \global\@minicapactivefalse%        Flag für nächste Gruppe zurücksetzen
}
\makeatother
