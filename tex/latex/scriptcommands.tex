\ExplSyntaxOn
\keys_define:nn { schooltex / AB }
  {
    mode .choices:nn = { small, smalllandscape, landscape, default }
      { \tl_set:Nn \l_school_mode_tl { #1 } },
    mode .initial:n = default,
    % Hier könnten weitere Keys wie 'pages' oder 'frame' hin
  }

\ExplSyntaxOn
\keys_define:nn { schooltex / AB }
  {
    % Wir definieren Schalter, die intern eine Hilfsvariable setzen
    small .meta:n = { mode = small },
    smalllandscape .meta:n = { mode = smalllandscape },
    landscape .meta:n = { mode = landscape },
    default .meta:n = { mode = default },
    
    % Das eigentliche Speicher-Feld
    mode .tl_set:N = \l_school_mode_tl,
    mode .initial:n = default,
  }

\NewDocumentCommand{\AB}{ O{default} m }{
  \group_begin:
  % Jetzt versteht er [small] direkt
  \keys_set:nn { schooltex / AB } { #1 }
  
  \phantomsection
  \label{pdf:#2}
  
  \cs_if_exist:cTF { Path } {
    \str_case:VnF \l_school_mode_tl {
      { small }          { \includepdf[pages=-, nup=1x2, doublepages, landscape, frame]{\Path/#2/#2} }
      { smalllandscape }  { \includepdf[pages=-, nup=1x2, doublepages, frame]{\Path/#2/#2} }
      { landscape }       { \includepdf[pages=-, landscape]{\Path/#2/#2} }
    }
    { \includepdf[pages=-]{\Path/#2/#2} } % Default-Fall (F-Zweig von str_case)
    
    \addcontentsline{toc}{subsection}{#2}
  }{
    \PackageError{schooltex}{Path not set.}{Setze \string\Path.}
  }
  \group_end:
}

% --- Gemeinsames Styling für die Abstände ---
\newcommand{\headingvspacing}{\par}

% Dieser Befehl sorgt dafür, dass die Zeile NACH der Überschrift NICHT eingerückt wird
\newcommand{\headingafter}{%
  \par\nopagebreak\vspace{0.8em}%
  \@afterindentfalse%
  \@afterheading%
}

% 1. Kapitel: \kap und \kap*
\NewDocumentCommand{\kap}{s m}{
  \headingvspacing
  \IfBooleanTF{#1}{
    {\scshape\large \ul{\mbox{#2}}}
  }{
    \stepcounter{kap}
    \noindent{\scshape\large \thekap\hspace{0.7em}\mbox{\ul{#2}}}
  }
  \headingafter
}

% 2. Unterkapitel: \uber und \uber*
\NewDocumentCommand{\uber}{s m}{
  \headingvspacing
  \IfBooleanTF{#1}{
    \noindent\ul{\mbox{#2}}
  }{
    \refstepcounter{unterkap}
    \noindent\ul{\mbox{\thekap.\theunterkap\hspace{0.7em}#2}}
  }
  \headingafter
}

% 3. Unter-Unterkapitel: \uuber und \uuber*
\NewDocumentCommand{\uuber}{s m}{
  \headingvspacing
  \IfBooleanTF{#1}{
    \noindent{\small \ul{\mbox{#2}}}
  }{
    \refstepcounter{unterunterkap}
    \noindent{\mbox{\small \thekap.\theunterkap.\theunterunterkap\hspace{0.7em}\ul{#2}}}
  }
  \headingafter
}

\makeatother
