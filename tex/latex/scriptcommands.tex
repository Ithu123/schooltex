% --- Gemeinsames Styling für die Abstände ---
% Vorher: ca 1.5em, Nachher: 1em. Verhindert Seitenumbruch nach der Zeile.
\newcommand{\headingvspacing}{\par\addvspace{1.5em}\noindent}
\newcommand{\headingafter}{\par\nopagebreak\vspace{0.8em}\noindent}

% 1. Kapitel (KAP)
\newcommand{\kap}[1]{
  \headingvspacing
  \stepcounter{kap}
  {\scshape\large \thekap\hspace{0.7em}\underline{#1}}
  \headingafter
}

% 2. Unterkapitel (UBER)
\newcommand{\uber}[1]{
  \headingvspacing
  \refstepcounter{unterkap}
  {\thekap.\theunterkap\hspace{0.7em}\underline{#1}}
  \headingafter
}

% 3. Unter-Unterkapitel (UUBER)
\newcommand{\uuber}[1]{
  \headingvspacing
  \refstepcounter{unterunterkap}
  {\small \thekap.\theunterkap.\theunterunterkap\hspace{0.7em}\underline{#1}}
  \headingafter
}
